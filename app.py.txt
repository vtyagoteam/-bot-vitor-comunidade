from flask import Flask, request
import requests
import os

app = Flask(__name__)

# --- SEUS DADOS ---
TOKEN = os.environ.get("TOKEN") # você vai colocar no Render
ID_NUMERO = os.environ.get("ID_NUMERO")
LINK_COMUNIDADE = "https://chat.whatsapp.com/JPvLwPHZcc42v8JDdHgqAe"
# ------------------

def enviar_msg(para, texto):
    url = f"https://graph.facebook.com/v20.0/{ID_NUMERO}/messages"
    headers = {"Authorization": f"Bearer {TOKEN}", "Content-Type": "application/json"}
    data = {
        "messaging_product": "whatsapp",
        "to": para,
        "text": {"body": texto}
    }
    requests.post(url, headers=headers, json=data)

@app.route("/", methods=["GET"])
def home():

@app.route("/webhook", methods=["GET", "POST"])
def webhook():
    if request.method == "GET":
        # Validação da Meta
        if request.args.get("hub.verify_token") == "vitor123":
            return request.args.get("hub.challenge")
        return "Erro de validação"

    # Recebe mensagem
    try:
        entry = request.json['entry'][0]['changes'][0]['value']
        if 'messages' in entry:
            msg = entry['messages'][0]
            de = msg['from']
            texto = msg.get('text', {}).get('body', '').lower()

            if texto in ["oi", "olá", "ola", "menu", "inicio"]:
                resposta = f"Olá! 👋 Aqui é do Vitor!\n\nTenho uma Comunidade exclusiva com ofertas, novidades e avisos em primeira mão.\n\n👉 Digite:\n1️⃣ - Quero entrar na Comunidade\n2️⃣ - Falar com o Vitor"
            elif texto == "1":
                resposta = f"Boa! Entra aqui e já aproveita: 👇\n{LINK_COMUNIDADE}\n\nTe espero lá dentro! 🚀"
            elif texto == "2":
                resposta = "Perfeito, já te atendo! Me diz o que você precisa? 😊"
            else:
                resposta = f"Não entendi. Digite *oi* para ver o menu.\nOu entre direto na comunidade: {LINK_COMUNIDADE}"

            enviar_msg(de, resposta)
    except Exception as e:
        print(e)
    return "ok", 200
